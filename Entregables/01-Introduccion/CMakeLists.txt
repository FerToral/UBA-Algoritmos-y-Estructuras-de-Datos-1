cmake_minimum_required(VERSION 3.13)
project(EntregablesIntroduccion LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_library(introduccion ejercicios.cpp)
foreach(ejercicio fibonacci division swap estaordenado rotar)
    add_executable(${ejercicio} ${ejercicio}_cli.cpp)
    target_link_libraries(${ejercicio} PRIVATE introduccion)
endforeach()
add_executable(introduccion_tests pruebas.cpp)
target_link_libraries(introduccion_tests PRIVATE introduccion)
add_test(NAME introduccion COMMAND introduccion_tests)
