cmake_minimum_required(VERSION 3.13)
project(Introduccion LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(introduccion ejercicios.cpp)
add_executable(introduccion_demo main.cpp)
target_link_libraries(introduccion_demo PRIVATE introduccion)
enable_testing()
add_executable(introduccion_pruebas pruebas.cpp)
target_link_libraries(introduccion_pruebas PRIVATE introduccion)
add_test(NAME introduccion COMMAND introduccion_pruebas)
