cmake_minimum_required(VERSION 3.13)
project(EntradaSalida LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(entrada_salida ejercicios.cpp)
add_executable(entrada_salida_demo main.cpp)
target_link_libraries(entrada_salida_demo PRIVATE entrada_salida)
enable_testing()
add_executable(entrada_salida_pruebas pruebas.cpp)
target_link_libraries(entrada_salida_pruebas PRIVATE entrada_salida)
add_test(NAME entrada_salida COMMAND entrada_salida_pruebas)
