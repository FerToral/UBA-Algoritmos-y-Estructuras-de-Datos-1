cmake_minimum_required(VERSION 3.13)
project(VectoresArchivos LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(vectores_archivos ejercicios.cpp)
add_executable(vectores_archivos_demo main.cpp)
target_link_libraries(vectores_archivos_demo PRIVATE vectores_archivos)
enable_testing()
add_executable(vectores_archivos_pruebas pruebas.cpp)
target_link_libraries(vectores_archivos_pruebas PRIVATE vectores_archivos)
add_test(NAME vectores_archivos COMMAND vectores_archivos_pruebas)
