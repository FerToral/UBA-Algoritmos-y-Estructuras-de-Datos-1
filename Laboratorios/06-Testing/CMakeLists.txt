cmake_minimum_required(VERSION 3.13)
project(LaboratorioTesting LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
set(BUILD_GTEST ON CACHE BOOL "" FORCE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../Dependencias/googletest" "${CMAKE_CURRENT_BINARY_DIR}/googletest")
add_library(ejercicios ej1/esPrimo.cpp ej1/puntaje.cpp ej2/llenarTaxis.cpp
    ej3/baldosasDelPiso.cpp ej3/contandoDragones.cpp ej3/electores.cpp)
target_include_directories(ejercicios PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(testing_tests pruebas/regresiones.cpp)
target_link_libraries(testing_tests PRIVATE ejercicios gtest_main)
add_test(NAME testing COMMAND testing_tests)
