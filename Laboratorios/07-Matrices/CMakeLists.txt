cmake_minimum_required(VERSION 3.13)
project(Matrices LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
set(BUILD_GTEST ON CACHE BOOL "" FORCE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../Dependencias/googletest" "${CMAKE_CURRENT_BINARY_DIR}/googletest")
add_library(ejercicios ejercicios.cpp)
target_include_directories(ejercicios PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(matrices_tests pruebas.cpp)
target_link_libraries(matrices_tests PRIVATE ejercicios gtest_main)
add_test(NAME matrices COMMAND matrices_tests)
