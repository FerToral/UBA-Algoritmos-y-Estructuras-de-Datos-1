cmake_minimum_required(VERSION 3.13)
project(Cadenas LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
set(BUILD_GTEST ON CACHE BOOL "" FORCE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../Dependencias/googletest" "${CMAKE_CURRENT_BINARY_DIR}/googletest")
add_library(cadenas cadenas.cpp src/risas.cpp)
add_executable(rotar rotar_cli.cpp)
add_executable(subsecuencia subsecuencia_cli.cpp)
target_link_libraries(rotar PRIVATE cadenas)
target_link_libraries(subsecuencia PRIVATE cadenas)
add_executable(cadenas_tests pruebas.cpp)
target_compile_options(cadenas_tests PRIVATE $<$<CXX_COMPILER_ID:GNU>:-include> $<$<CXX_COMPILER_ID:GNU>:sstream>)
target_link_libraries(cadenas_tests PRIVATE cadenas gtest_main)
add_test(NAME cadenas COMMAND cadenas_tests)
set_tests_properties(cadenas PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
