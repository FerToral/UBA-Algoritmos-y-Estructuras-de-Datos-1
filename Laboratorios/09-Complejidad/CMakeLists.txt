cmake_minimum_required(VERSION 3.13)
project(Complejidad LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(complejidad ejercicios.cpp)
add_executable(complejidad_demo main.cpp)
target_link_libraries(complejidad_demo PRIVATE complejidad)
enable_testing()
add_executable(complejidad_pruebas pruebas.cpp)
target_link_libraries(complejidad_pruebas PRIVATE complejidad)
add_test(NAME complejidad COMMAND complejidad_pruebas)
