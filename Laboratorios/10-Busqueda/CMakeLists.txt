cmake_minimum_required(VERSION 3.13)
project(Busqueda LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(busqueda ejercicios.cpp)
add_executable(busqueda_demo main.cpp)
target_link_libraries(busqueda_demo PRIVATE busqueda)
enable_testing()
add_executable(busqueda_pruebas pruebas.cpp)
target_link_libraries(busqueda_pruebas PRIVATE busqueda)
add_test(NAME busqueda COMMAND busqueda_pruebas)
