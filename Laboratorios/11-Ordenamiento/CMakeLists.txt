cmake_minimum_required(VERSION 3.13)
project(Ordenamiento LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ordenamiento ejercicios.cpp)
add_executable(ordenamiento_demo main.cpp)
target_link_libraries(ordenamiento_demo PRIVATE ordenamiento)
enable_testing()
add_executable(ordenamiento_pruebas pruebas.cpp)
target_link_libraries(ordenamiento_pruebas PRIVATE ordenamiento)
add_test(NAME ordenamiento COMMAND ordenamiento_pruebas)
